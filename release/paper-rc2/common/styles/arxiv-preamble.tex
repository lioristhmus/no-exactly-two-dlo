% U+220E END OF PROOF: render a small hollow square at the right margin.
\newcommand{\rightqedsquare}{%
  \ifhmode
    \unskip\nobreak\hfill\ensuremath{\square}%
  \else
    \noindent\hfill\ensuremath{\square}%
  \fi
}
\newunicodechar{∎}{\rightqedsquare}
